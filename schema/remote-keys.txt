# Schema keys this Garmin remote depends on — the "Monkey C mirror" of
# faBolus/schema/command.schema.json (the source of truth). The remote consumes a SUBSET of the
# contract (it's units-only and has no double-confirm-token flow), so this lists only the keys it
# actually reads or writes, not every schema property.
#
# scripts/check-schema-drift.sh asserts each key here (a) still exists in the schema and (b) is still
# referenced in the Monkey C source (source/app/RemoteComm.mc / AppState.mc). Renaming a schema key,
# or dropping one from the code without updating this list, fails CI. One key per line; # = comment.

# Envelope (every message)
version
kind
requestId

# Outbound commands (RemoteComm.mc)
units
carbsGrams
remoteEstimateUnits
sentAt
forceGlucose
alertId
alertKind
# C2 §2.3: the entered 4-digit bolus passcode (String), added to bolusRequest only when a code was
# entered. The phone verifies it and denies the bolus if wrong/absent; the watch never verifies/persists.
bolusPasscode
# AB4 (Addendum B Option B): explicit per-attempt include-stale INTENT (Boolean), added to a CARB
# bolusRequest only when the wearer chose "include" this attempt. Absent ⇒ host fails closed to carbs-only.
includeStaleBG

# Phase 20 (F1, D-02): which pump-status fields fill the three user-assignable complication slots
# (ordered, <=3). Parsed + sanitized in AppState.handle (garminComplicationSlots).
garminComplicationSlots
# Phase 20 (R1/R4/F3, D-01): the phone-owned alert-intensity setting mirrored to the watch's fail-closed
# gate (AppState.handle parse + alertActionFor). Absent ⇒ watch defaults to vibration-only.
alertIntensityMode
alertAudibleMinSeverity
alertCriticalOverridesDnd
# Phase 20 (D-01): each `alerts` entry also carries an optional per-alert `severity` tier
# (AppState.sanitizeAlerts preserves it; alertSeverityTier fails an absent one closed to "critical").
# NOT a separate top-level key — it is a nested property of the `alerts` array item above (the
# schema-drift check tracks only top-level keys, like id/kind/title, which `alerts` already covers).
remotesReadOnly
canBolus
bolusBlockReason
supportsRemoteAlertDismiss
# CX-G-08 (14-09) — DYNAMIC pump-tied capability: true only when the phone build supports the
# authenticated dismissAck path AND the connected pump honors a remote dismiss. Persisted (mirrors
# garminBolusEnabled), parsed BEFORE the alerts replace (H1).
supportsDismissAck
# 14-10 — DYNAMIC pump-tied capability, the exact negation of supportsDismissAck: the raw-snapshot
# proof-of-absence backstop for a pump that does NOT honor a remote dismiss (t:slim X2). Persisted
# (mirrors supportsDismissAck), parsed BEFORE the alerts replace (same H1 reason).
supportsRawAlertSnapshot
# 14-10 — the pump's TRUE pre-local-snooze-filter active-alert bitmap for this poll, emitted only when
# supportsRawAlertSnapshot is true AND the host's raw set is known. A reconciliation oracle for the
# wearer's own pending dismissals only — never a display source.
rawAlerts
garminBolusEnabled
bolusPasscodeRequired
controllerVariant
controlIQEnabled
